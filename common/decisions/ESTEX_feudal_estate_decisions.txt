# The front door of the mod: a feudal ruler founds their family seat. The estate
# itself comes from the noble family title below, change_government on its own
# never creates estate domiciles (camps and yurts yes, estates no, they hang off
# the title). All the plumbing is hidden, the player just sees the summary and
# gets the founding event.
establish_feudal_estate_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_castle_view.dds"
	}
	decision_group_type = major
	ai_check_interval = 0 # AI off for now, team decision pending

	desc = establish_feudal_estate_decision_desc
	selection_tooltip = establish_feudal_estate_decision_tooltip
	confirm_text = establish_feudal_estate_decision_confirm

	cost = {
		gold = 150
		prestige = 100
	}

	is_shown = {
		government_has_flag = government_is_feudal
		is_landed = yes
		trigger_if = {
			limit = { government_has_flag = government_is_feudal_estate }
			# recovery for pre-succession-fix saves: an heir stuck in our
			# government without the estate title gets to refound
			NOT = { any_held_title = { has_variable = feudal_estate_title } }
		}
	}

	is_valid = {
		is_available_adult = yes
		is_at_war = no
	}

	effect = {
		custom_tooltip = establish_feudal_estate_effect_tt
		hidden_effect = {
			# recovery path (pre-succession-fix saves): the ruler is stuck in the
			# estate government without the estate title. If a landless relative
			# still holds the old title, destroy it so the house gets a clean
			# slate, then drop to plain feudal - the title creation below refuses
			# to run when the holder already has the government we pass in.
			if = {
				limit = { government_has_flag = government_is_feudal_estate }
				house ?= {
					random_house_member = {
						limit = {
							NOT = { this = root }
							is_ai = yes
							is_landed = no
							any_held_title = { has_variable = feudal_estate_title }
						}
						random_held_title = {
							limit = { has_variable = feudal_estate_title }
							save_scope_as = fe_stray_title
						}
					}
				}
				if = {
					limit = { exists = scope:fe_stray_title }
					destroy_title = scope:fe_stray_title
				}
				change_government = feudal_government
			}
			# noble family titles can only go to house heads, so sort that out first.
			# Same logic as create_noble_family_effect: grab headship off a less
			# important AI head, otherwise split off a cadet branch.
			if = {
				limit = {
					exists = house.house_head
					NOT = { house.house_head = root }
				}
				if = {
					limit = {
						house.house_head ?= {
							is_ai = yes # Don't steal house head from a player
							OR = {
								# Within same realm but less important
								AND = {
									top_liege = root.top_liege
									OR = {
										is_ruler = no
										highest_held_title_tier < root.highest_held_title_tier
									}
								}
								# Wanderer
								NOT = { exists = liege }
							}
						}
					}
					house = { set_house_head = root }
				}
				else = {
					create_cadet_branch = {}
				}
			}
			if = {
				limit = { has_domicile = no }
				# NOTE: this has to run before change_government, the title creation
				# fails if the holder already has the government we pass in here.
				# Tier matches the holder so counts don't get bumped up a rank by a
				# landless duchy title.
				if = {
					limit = { highest_held_title_tier >= tier_duchy }
					give_noble_family_title = {
						name = noble_family_name
						tier = duchy
						article = DEFAULT_TITLE_NAME_ARTICLE
						government = feudal_estate_government
						save_scope_as = feudal_estate_title
					}
				}
				else = {
					give_noble_family_title = {
						name = noble_family_name
						tier = county
						article = DEFAULT_TITLE_NAME_ARTICLE
						government = feudal_estate_government
						save_scope_as = feudal_estate_title
					}
				}
				# mark it as ours so the on_actions can find it (succession + cleanup
				# on land loss)
				scope:feudal_estate_title ?= {
					set_variable = {
						name = feudal_estate_title
						value = yes
					}
					# swap out the engine-assigned house-head succession so the
					# estate follows the realm heir (and displays them as heir)
					add_title_law = estex_feudal_estate_succession_law
				}
			}
			change_government = feudal_estate_government
			domicile ?= {
				move_domicile = root.capital_province
				set_up_domicile_estate_effect = yes
			}
			# the founding ceremony
			trigger_event = {
				id = estex_estate.4001
				days = 1
			}
		}
	}

	ai_will_do = {
		base = 0
	}
}
