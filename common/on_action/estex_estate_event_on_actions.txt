# Event pulse for estate flavor events. Hooked through the on_actions list which
# merges between files, same reason as the title on_actions (an effect block here
# would overwrite vanilla's).

yearly_playable_pulse = {
	on_actions = { estex_estate_yearly_pulse }
}

estex_estate_yearly_pulse = {
	trigger = {
		government_has_flag = government_is_feudal_estate
		exists = domicile
		domicile = { is_domicile_type = estex_estate }
	}
	random_events = {
		chance_of_no_event = {
			value = 60
		}
		# castellan
		100 = estex_estate.1001
		100 = estex_estate.1002
		50 = estex_estate.1003
		50 = estex_estate.1010
		25 = estex_estate.1011
		# deer park
		100 = estex_estate.2001
		20 = estex_estate.2002
		75 = estex_estate.2003
		75 = estex_estate.2004
		# tournament grounds
		75 = estex_estate.3001
		75 = estex_estate.3002
		50 = estex_estate.3003
		50 = estex_estate.3004
		50 = estex_estate.3005
		15 = estex_estate.3006
		# cells and dungeon
		75 = estex_estate.3101
		75 = estex_estate.3102
		25 = estex_estate.3103
		75 = estex_estate.3104
		50 = estex_estate.3105
		50 = estex_estate.3106
		# crossovers
		50 = estex_estate.3201
		50 = estex_estate.3202
	}
}
