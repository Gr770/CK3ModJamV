estex_enjoy_luxurious_bath_decision_effect = {
	if = {
		limit = {
			domicile ?= { has_domicile_building_or_higher = estex_bath_04 }
		}
		stress_impact = {
			base = major_stress_impact_loss
			lazy = minor_stress_impact_loss
			content = minor_stress_impact_loss
			calm = minor_stress_impact_loss
		}	
	}
	else_if = {
		limit = {
			domicile ?= { has_domicile_building_or_higher = estex_bath_03 }
		}
		stress_impact = {
			base = medium_stress_impact_loss
			lazy = minor_stress_impact_loss
			content = minor_stress_impact_loss
			calm = minor_stress_impact_loss
		}	
	}
	else_if = {
		limit = {
			domicile ?= { has_domicile_building_or_higher = estex_bath_02 }
		}
		stress_impact = {
			base = minor_stress_impact_loss
			lazy = miniscule_stress_impact_loss
			content = miniscule_stress_impact_loss
			calm = miniscule_stress_impact_loss
		}	
	}
	else = {
		stress_impact = {
			base = miniscule_stress_impact_loss
			lazy = miniscule_stress_impact_loss
			content = miniscule_stress_impact_loss
			calm = miniscule_stress_impact_loss
		}		
	}
}
