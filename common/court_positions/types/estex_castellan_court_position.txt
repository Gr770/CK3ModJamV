# Castellan, the keeper of the estate. First of our building unlocked court
# positions: shows up for estate government rulers and can only be filled once
# the keep is stone (main_02). Seneschal runs the household, castellan runs the
# castle, they shouldn't step on each other.
estex_castellan_court_position = {
	sort_order = 298
	max_available_positions = 1
	skill = stewardship

	court_position_asset = {
		animation = reading
		background = "gfx/interface/illustrations/event_scenes/ep3_medi_estate.dds"
	}

	opinion = {
		value = regular_court_position_opinion
	}

	aptitude_level_breakpoints = { 20 40 60 80 }

	aptitude = {
		value = 25
		add = {
			value = stewardship
			multiply = 2.5
			max = 50
			desc = court_position_skill_stewardship
		}
		if = {
			limit = {
				has_trait = administrator
			}
			add = {
				value = 20
				desc = court_position_administrator_trait
			}
		}
		if = {
			limit = {
				has_trait = overseer
			}
			add = {
				value = 20
				desc = court_position_overseer_trait
			}
		}
		if = {
			limit = {
				has_trait = architect
			}
			add = {
				value = 20
				desc = court_position_architect_trait
			}
		}
		# a grander keep attracts (and trains) a better castellan
		add = {
			value = 0
			if = {
				limit = {
					employer.domicile ?= {
						has_domicile_building_or_higher = estex_estate_main_04
					}
				}
				add = 20
			}
			else_if = {
				limit = {
					employer.domicile ?= {
						has_domicile_building_or_higher = estex_estate_main_03
					}
				}
				add = 10
			}
			desc = estex_castellan_keep_aptitude
		}
		add = court_position_aptitude_low_penalty_value
		add = general_aptitude_bonus_exclude_other_court_position
	}

	is_shown = {
		government_has_flag = government_is_feudal_estate
		exists = domicile
		domicile = { is_domicile_type = estex_estate }
	}

	# root is the court owner
	valid_position = {
		government_has_flag = government_is_feudal_estate
		custom_tooltip = {
			text = estex_castellan_needs_stone_keep_tt
			domicile ?= { has_domicile_building_or_higher = estex_estate_main_02 }
		}
	}

	is_shown_character = {
		scope:employee = {
			court_position_employee_shown_trigger = yes
		}
	}

	valid_character = {
		scope:employee = {
			estex_castellan_validity_trigger = { EMPLOYER = scope:liege }
		}
	}

	revoke_cost = {
		prestige = minor_court_position_prestige_revoke_cost
	}

	salary = {
		gold = medium_court_position_salary
		prestige = minor_court_position_prestige_salary
		round = no
	}

	received_salary = {
		gold = {
			value = medium_court_position_salary
		}
		prestige = {
			value = minor_court_position_prestige_salary
		}
		round = no
	}

	scaling_employer_modifiers = {
		terrible = {
			domicile_monthly_gold_add = 0.1
		}
		poor = {
			domicile_monthly_gold_add = 0.15
			enemy_hostile_scheme_success_chance_add = -1
		}
		average = {
			domicile_monthly_gold_add = 0.2
			enemy_hostile_scheme_success_chance_add = -1
		}
		good = {
			domicile_monthly_gold_add = 0.3
			enemy_hostile_scheme_success_chance_add = -2
		}
		excellent = {
			domicile_monthly_gold_add = 0.5
			enemy_hostile_scheme_success_chance_add = -3
		}
	}

	modifier = {
		general_opinion = 5
		monthly_stewardship_lifestyle_xp_gain_mult = 0.1
	}

	on_court_position_vacated = {
		court_position_vacated_event_effect = {
			COURT_POSITION_TYPE = estex_castellan_court_position
		}
	}

	ai_position_score = {
		value = 100
	}

	ai_candidate_score = {
		value = 50
		add = scope:base_value
		add = court_position_candidate_score_base_value
		add = court_position_candidate_aptitude_value
	}
}
