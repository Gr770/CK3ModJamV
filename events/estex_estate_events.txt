# White hart chase loop: press on or give up
estex_estate.2007 = {
	type = character_event
	title = estex_estate.2007.t
	desc = {
		desc = estex_estate.2007.desc
		desc = estex_source_deer_park
	}
	theme = hunting

	left_portrait = {
		character = root
		animation = horse_archer_aggressive
	}

	option = { # Spur on
		name = estex_estate.2007.a
		custom_tooltip = estex_estate.2007.a.tt
		# every round in the saddle hones the craft
		if = {
			limit = { has_trait = lifestyle_hunter }
			add_trait_xp = {
				trait = lifestyle_hunter
				track = hunter
				value = 2
			}
		}
		else = {
			add_trait = lifestyle_hunter
		}
		random_list = {
			40 = { # The kill
				modifier = {
					add = 20
					prowess >= 8
				}
				modifier = {
					add = 15
					domicile ?= { has_domicile_parameter = estex_estate_has_chase }
				}
				# the beast tires as the chase drags on
				modifier = {
					add = 10
					var:estex_hart_chase_rounds >= 1
				}
				modifier = {
					add = 10
					var:estex_hart_chase_rounds >= 2
				}
				custom_tooltip = estex_estate.2007.success.tt
				trigger_event = {
					id = estex_estate.2005
					days = 1
				}
			}
			60 = { # It slips away again
				custom_tooltip = estex_estate.2007.fail.tt
				add_stress = 10
				change_variable = {
					name = estex_hart_chase_rounds
					add = 1
				}
				if = { # three failed rounds and it is gone for good
					limit = {
						var:estex_hart_chase_rounds >= 3
					}
					trigger_event = {
						id = estex_estate.2006
						days = 1
					}
				}
				else = {
					trigger_event = {
						id = estex_estate.2007
						days = 2
					}
				}
			}
		}
		ai_chance = { base = 60 }
	}
	option = { # Abandon the chase
		name = estex_estate.2007.b
		remove_variable = estex_hart_chase_rounds
		ai_chance = { base = 40 }
	}
}

# White hart follow up: the kill. The antler trinket and the prestige are handed
# out in immediate so the options stay clean, the choice is just about the meat.
estex_estate.2005 = {
	type = character_event
	title = estex_estate.2005.t
	desc = {
		desc = estex_estate.2005.desc
		desc = estex_source_deer_park
	}
	theme = hunting

	immediate = {
		if = {
			limit = { has_variable = estex_hart_chase_rounds }
			remove_variable = estex_hart_chase_rounds
		}
		add_prestige = 100
		# a legendary kill makes a hunter of anyone, vanilla pattern from the
		# legendary hunting lodge
		if = {
			limit = { has_trait = lifestyle_hunter }
			add_trait_xp = {
				trait = lifestyle_hunter
				track = hunter
				value = 15
			}
		}
		else = {
			add_trait = lifestyle_hunter
		}
		# antler trinket rather than a wall skull: personal artifact, works for
		# any rank, no royal court needed
		create_artifact_animal_trinket_effect = {
			OWNER = root
			HUNTER = root
			LEGENDARY = yes
			ANIMAL = flag:stag
		}
	}

	left_portrait = {
		character = root
		animation = celebrate_spear
	}

	artifact = {
		target = scope:newly_created_artifact
		position = lower_center_portrait
	}

	option = { # Keep the meat for a celebratory feast
		name = estex_estate.2005.a
		add_stress = minor_stress_loss
		ai_chance = { base = 60 }
	}
	option = { # Share the meat with the village
		name = estex_estate.2005.b
		add_piety = 25
		ai_chance = { base = 40 }
	}
}
