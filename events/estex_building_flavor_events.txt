# Character inspects the kitchen

	type = character_event
	title = estex_building_flavor.0100.t
	desc = estex_building_flavor.0100.desc
	theme = family
	override_background = { reference = bp1_kitchen_western }

	left_portrait = {
		character = root
		animation = survey
	}
	
	option = { # Begin the inspection!
		name = estex_building_flavor.0100.a
		random_list = {
			10 = { # Nothing out of the ordinary

					domicile ?= { has_domicile_building_or_higher = estex_kitchen_02 }	
					add = 5
				}

					domicile ?= { has_domicile_building_or_higher = estex_kitchen_03 }	
					add = 5
				}

			10 = { # Very bad!
				desc = estex_building_flavor.0100.toast.3
				show_chance = no

# Character practices hunting
estex_building_flavor.0110 = {
	type = character_event
	title = estex_building_flavor.0110.t
	desc = estex_building_flavor.0110.desc
	theme = hunting

	left_portrait = {
		character = root
		triggered_animation = {
			trigger = {
				culture_has_central_asian_heritage_pillar_trigger = yes
			}
			animation = hunting_shortbow_aim_arrow_default
		}
		animation = hunting_knife_start
	}
	
	option = { # Begin the inspection!
		name = estex_building_flavor.0110.a
		estex_train_hunting_decision_effect = yes
	}
}

# Character enjoys luxurious bath
estex_building_flavor.0120 = {
	type = character_event
	title = estex_building_flavor.0120.t
	desc = estex_building_flavor.0120.desc
	theme = friendly
	override_background  = { reference = bath_house }

	left_portrait = {
		character = root
		animation = happiness
	}

	immediate = {
		add_character_flag = is_naked
	}
	
	option = { # ok
		name = estex_building_flavor.0120.a
		estex_enjoy_luxurious_bath_decision_effect = yes
	}

	after = {
		if = {
			limit = { has_character_flag = is_naked }
			remove_character_flag = is_naked
		}		
	}

}
